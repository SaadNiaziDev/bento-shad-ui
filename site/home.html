React

import 'bento-shad-ui/bento.css';
import { BentoGrid, Tile, Stat, Button, toast } from 'bento-shad-ui/react';

export function Overview() {
  return (
    <BentoGrid>
      <Tile span={2} rows={2} tone="inverse">
        <Button variant="brand" onClick={() => toast('Saved')}>Save</Button>
      </Tile>
      <Tile tone="peach"><Stat label="Bookings" value="1,284" /></Tile>
    </BentoGrid>
  );
}
